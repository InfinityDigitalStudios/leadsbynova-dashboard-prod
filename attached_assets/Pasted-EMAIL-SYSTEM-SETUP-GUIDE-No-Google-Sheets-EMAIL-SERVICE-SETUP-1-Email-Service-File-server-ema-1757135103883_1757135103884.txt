// POST /api/lead-complete
app.post('/api/lead-complete', async (req, res) => {
  try {
    const { formData, chatData } = req.body;
    
    // Prepare consolidated data
    const consolidatedData = {
      // Form data
      formTimestamp: formData.timestamp,
      name: formData.name,
      email: formData.email,
      phone: formData.phone,
      guide_type: formData.guide_type,
      
      // Chat data
      chatTimestamp: chatData.timestamp,
      prequalified: chatData.prequalified,
      preQualificationRange: chatData.preQualificationRange,
      moveTimeline: chatData.moveTimeline,
      haveAgent: chatData.haveAgent,
      budgetRange: chatData.budgetRange,
      propertyType: chatData.propertyType,
      bookedCall: chatData.bookedCall,
      daySelected: chatData.daySelected,
      
      // Status
      completedChat: true
    };
    
    // Send agent notification
    await emailService.sendConsolidatedNotification(consolidatedData);
    console.log(`Agent notification sent for: ${formData.name}`);
    
    // Send client welcome email
    const isCallBooked = chatData.bookedCall?.includes('Yes') && chatData.bookedCall?.includes('schedule');
    await emailService.sendWelcomeEmail({
      name: formData.name,
      email: formData.email,
      guide_type: formData.guide_type,
      scheduledCall: isCallBooked ? chatData.daySelected : undefined
    });
    console.log(`Client welcome email sent to: ${formData.email}`);
    
    res.json({ success: true, message: 'Emails sent successfully' });
    
  } catch (error) {
    console.error('Email sending error:', error);
    res.status(500).json({ error: 'Failed to send emails' });
  }
});
2. Form Submission Only (Chat Timeout)
// POST /api/lead-timeout
app.post('/api/lead-timeout', async (req, res) => {
  try {
    const { formData } = req.body;
    
    // Prepare data for incomplete chat
    const consolidatedData = {
      // Form data
      formTimestamp: formData.timestamp,
      name: formData.name,
      email: formData.email,
      phone: formData.phone,
      guide_type: formData.guide_type,
      
      // Status
      completedChat: false
    };
    
    // Send agent notification (form only)
    await emailService.sendConsolidatedNotification(consolidatedData);
    console.log(`Agent timeout notification sent for: ${formData.name}`);
    
    // Send client welcome email (generic)
    await emailService.sendWelcomeEmail({
      name: formData.name,
      email: formData.email,
      guide_type: formData.guide_type,
      scheduledCall: undefined // No appointment since chat wasn't completed
    });
    console.log(`Client welcome email sent to: ${formData.email} (generic)`);
    
    res.json({ success: true, message: 'Timeout emails sent successfully' });
    
  } catch (error) {
    console.error('Timeout email error:', error);
    res.status(500).json({ error: 'Failed to send timeout emails' });
  }
});
